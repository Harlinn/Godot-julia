using Godot

@enum Mode Idle Run

struct Stats
    hits::Int
end

@godot_class mutable struct Mover <: Node
    @export speed::Float64 = 1.0
    count::Int = 0
    label::String = "v1"
    mode::Mode = Idle
    stats::Stats = Stats(0)
    saved::Int = 0
    @signal moved(n::Int)
end

version(self::Mover) = 1

# Hot-reload hook (C#: ISerializationListener.OnBeforeSerialize).
Godot._before_reload(self::Mover) = (self.saved = 100 * self.count; nothing)

function step(self::Mover)
    self.count += 1
    self.mode = Run
    self.stats = Stats(self.stats.hits + 1)
    return self.count
end

arm(self::Mover) = (connect(self, :moved, n -> (self.label = "v1:$n"; nothing)); nothing)
